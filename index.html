const taskInput = document.getElementById('taskInput'); const taskTime = document.getElementById('taskTime'); const addBtn = document.getElementById('addBtn'); const memoList = document.getElementById('memoList'); const swooshSound = document.getElementById('swooshSound'); const clearOldBtn = document.getElementById('clearOldBtn'); // 1. 初始化：從 localStorage 讀取舊資料 let savedMemos = JSON.parse(localStorage.getItem('myMemos')) || []; renderMemos(); // 預設時間設為現在 const now = new Date(); now.setMinutes(now.getMinutes() - now.getTimezoneOffset()); taskTime.value = now.toISOString().slice(0, 16); // 2. 渲染清單的函式 function renderMemos() { memoList.innerHTML = ''; // 按時間排序 savedMemos.sort((a, b) => new Date(a.time) - new Date(b.time)); savedMemos.forEach((memo, index) => { const li = document.createElement('li'); li.className = 'memo-item'; li.dataset.index = index; li.dataset.time = memo.time; li.innerHTML = ` ${memo.time.replace('T', ' ')} ${memo.content} `; addSwipeListener(li, index); memoList.appendChild(li); }); } // 3. 新增功能（包含儲存） addBtn.addEventListener('click', () => { if (!taskInput.value) return; const newMemo = { content: taskInput.value, time: taskTime.value, id: Date.now() // 給每筆資料一個唯一 ID }; savedMemos.push(newMemo); saveToLocal(); // 存入 localStorage renderMemos(); taskInput.value = ''; }); // 4. 側滑偵測邏輯 function addSwipeListener(el, index) { let startX = 0; el.addEventListener('touchstart', (e) => startX = e.touches[0].clientX); el.addEventListener('touchmove', (e) => { let moveX = e.touches[0].clientX - startX; if (moveX < 0) el.style.transform = `translateX(${moveX}px)`; }); el.addEventListener('touchend', (e) => { let moveX = e.changedTouches[0].clientX - startX; if (moveX < -150) { triggerDelete(el, index); } else { el.style.transform = `translateX(0)`; } }); } // 5. 執行刪除與更新快取 function triggerDelete(el, index) { el.classList.add('fly-away'); swooshSound.currentTime = 0; swooshSound.play().catch(() => {}); setTimeout(() => { // 從陣列中移除該項目 savedMemos = savedMemos.filter((_, i) => i !== index); saveToLocal(); renderMemos(); // 重新渲染確保索引正確 }, 600); } // 6. 快速清理過期事項 clearOldBtn.addEventListener('click', () => { const nowTime = new Date().toISOString().slice(0, 16); const toDelete = savedMemos.filter(memo => memo.time < nowTime); if (toDelete.length === 0) return; // 找到所有過期的 DOM 元素並執行飛出動畫 const items = document.querySelectorAll('.memo-item'); items.forEach(item => { if (item.dataset.time < nowTime) { item.classList.add('fly-away'); } }); swooshSound.play().catch(() => {}); setTimeout(() => { savedMemos = savedMemos.filter(memo => memo.time >= nowTime); saveToLocal(); renderMemos(); }, 600); }); // 儲存到 LocalStorage 的工具 function saveToLocal() { localStorage.setItem('myMemos', JSON.stringify(savedMemos)); }
